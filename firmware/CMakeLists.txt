cmake_minimum_required(VERSION 3.16)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

idf_build_set_property(MINIMAL_BUILD ON)
option(CODEPULSE_AUDIO_ENABLED "Optional new-board speaker support (not legacy buzzer)" OFF)
option(CODEPULSE_AUDIO_INPUT_ENABLED "Optional PDM microphone and dedicated talk button" OFF)
option(CODEPULSE_AUDIO_PINS_CONFIRMED "New-board audio netlist and supply physically verified" OFF)
option(CODEPULSE_AUDIO_COMPILE_TEST "Compile audio but forbid hardware initialization" OFF)
option(CODEPULSE_AUDIO_AMP_ACTIVE_HIGH "Amplifier enable polarity" ON)
foreach(pin BCLK WS DOUT AMP_EN MIC_CLK MIC_DATA TALK_BUTTON)
    set(CODEPULSE_AUDIO_${pin} "-1" CACHE STRING "Confirmed new-board audio GPIO; -1 means not wired")
endforeach()
if(CODEPULSE_AUDIO_INPUT_ENABLED AND NOT CODEPULSE_AUDIO_ENABLED)
    message(FATAL_ERROR "Microphone requires the optional audio profile")
endif()
if(CODEPULSE_AUDIO_ENABLED AND NOT CODEPULSE_AUDIO_PINS_CONFIRMED AND NOT CODEPULSE_AUDIO_COMPILE_TEST)
    message(FATAL_ERROR "Audio pin map is unverified. Use COMPILE_TEST only until the new PCB is checked.")
endif()

option(CODEPULSE_QUIET_RECT_TRIAL "Opt-in byte-aligned narrow BW refresh windows" OFF)
option(CODEPULSE_QUIET_PAIR150_TRIAL "Opt-in live dashboard PAIR150 duration trial" OFF)
if((CODEPULSE_QUIET_PAIR150_TRIAL OR CODEPULSE_QUIET_RECT_TRIAL) AND (CODEPULSE_EPD_TEXT_TIMING_PROBE OR CODEPULSE_EPD_TEXT_PROBE OR CODEPULSE_EPD_QUIET_PROBE OR CODEPULSE_EPD_WAVEFORM_PROBE OR CODEPULSE_EPD_WINDOW_PROBE))
    message(FATAL_ERROR "PAIR150 live trial cannot be combined with manual probes")
endif()
option(CODEPULSE_EPD_WINDOW_PROBE "Build manual OTP window experiment (no sync)" OFF)
option(CODEPULSE_EPD_WAVEFORM_PROBE "Build manual register LUT A/B experiment" OFF)
option(CODEPULSE_EPD_QUIET_PROBE "Build manual BW old/new experiment" OFF)
option(CODEPULSE_EPD_TEXT_PROBE "Build bounded manual B-waveform text experiment" OFF)
option(CODEPULSE_EPD_TEXT_TIMING_PROBE "Build independent manual text timing experiment" OFF)
if(CODEPULSE_EPD_TEXT_TIMING_PROBE AND (CODEPULSE_EPD_TEXT_PROBE OR CODEPULSE_EPD_QUIET_PROBE OR CODEPULSE_EPD_WAVEFORM_PROBE OR CODEPULSE_EPD_WINDOW_PROBE))
    message(FATAL_ERROR "Choose only one panel experiment")
endif()
if((CODEPULSE_EPD_TEXT_PROBE AND (CODEPULSE_EPD_WINDOW_PROBE OR CODEPULSE_EPD_WAVEFORM_PROBE OR CODEPULSE_EPD_QUIET_PROBE)) OR
   (CODEPULSE_EPD_WAVEFORM_PROBE AND CODEPULSE_EPD_WINDOW_PROBE) OR
   (CODEPULSE_EPD_QUIET_PROBE AND (CODEPULSE_EPD_WAVEFORM_PROBE OR CODEPULSE_EPD_WINDOW_PROBE)))
    message(FATAL_ERROR "Choose only one panel experiment")
endif()
if(CODEPULSE_AUDIO_ENABLED AND CODEPULSE_AUDIO_COMPILE_TEST)
    set(PROJECT_VER "0.0.28-audio-check")
elseif(CODEPULSE_AUDIO_ENABLED)
    set(PROJECT_VER "0.0.28-audio")
elseif(CODEPULSE_EPD_TEXT_TIMING_PROBE)
    set(PROJECT_VER "0.0.8-text-cycle")
elseif(CODEPULSE_EPD_TEXT_PROBE)
    set(PROJECT_VER "0.0.5-text-probe")
elseif(CODEPULSE_EPD_QUIET_PROBE)
    set(PROJECT_VER "0.0.4-quiet-probe2")
elseif(CODEPULSE_EPD_WAVEFORM_PROBE)
    set(PROJECT_VER "0.0.3-waveform-probe2")
elseif(CODEPULSE_EPD_WINDOW_PROBE)
    set(PROJECT_VER "0.0.3-window-probe")
elseif(CODEPULSE_QUIET_RECT_TRIAL AND CODEPULSE_QUIET_PAIR150_TRIAL)
    set(PROJECT_VER "0.0.30-life-region-pair150")
elseif(CODEPULSE_QUIET_RECT_TRIAL)
    set(PROJECT_VER "0.0.30-life-region")
elseif(CODEPULSE_QUIET_PAIR150_TRIAL)
    set(PROJECT_VER "0.0.30-life-pair150")
else()
    set(PROJECT_VER "0.0.30-life")
endif()
project(codepulse_bringup)
